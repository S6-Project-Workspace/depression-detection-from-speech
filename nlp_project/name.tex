    \item \textbf{Solution}: Production-ready multimodal system combining audio, text, and linguistic analysis

    \begin{itemize}

% Dataset and Training
\begin{frame}
\frametitle{Dataset and Training}
\begin{itemize}
    \item \textbf{Dataset}: DravidianLangTech @ ACL 2026
    \begin{itemize}
        \item Tamil: 1,247 samples, Malayalam: 1,089 samples
        \item Binary classification: Depressed/Non-Depressed
        \item 16kHz audio with manual transcriptions
    \end{itemize}
    \item \textbf{Training Configuration}:
    \begin{itemize}
        \item 5-fold stratified cross-validation
        \item AdamW optimizer with cosine annealing
        \item Mixed precision training with gradient clipping
        \item Early stopping with validation monitoring
    \end{itemize}
\end{itemize}
\end{frame}
